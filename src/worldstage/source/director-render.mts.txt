function bounds(image:Image):Bounds{
  const c=createCanvas(image.width,image.height),x=c.getContext('2d');x.drawImage(image,0,0);
  const d=x.getImageData(0,0,image.width,image.height).data;
  let left=image.width,right=0,top=image.height,bottom=0,transparent=0,filled=0;
  for(let y=0;y<image.height;y++)for(let px=0;px<image.width;px++){
    const alpha=d[(y*image.width+px)*4+3];if(alpha===0)transparent++;
    if(alpha>12){filled++;left=Math.min(left,px);right=Math.max(right,px);top=Math.min(top,y);bottom=Math.max(bottom,y);}

  return{x:left,y:top,w:right-left+1,h:bottom-top+1,transparent,coverage:filled/(image.width*image.height)};
}
